🗄️ SQL की बुनियाद 
CREATE TABLE dost (naam TEXT, shahar TEXT, umra INTEGER);
INSERT INTO dost VALUES ('Raju', 'Delhi', 20);
SELECT * FROM dost;
SELECT naam FROM dost WHERE shahar = 'Delhi';
SELECT * FROM dost ORDER BY umra;
UPDATE dost SET umra = 21 WHERE naam = 'Raju';
DELETE FROM dost WHERE naam = 'Raju';
